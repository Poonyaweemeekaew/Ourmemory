        let selectedSituationId = null;

        function renderHandCards() {
            const container = document.getElementById('hand-cards-container');
            container.innerHTML = '';
            situationCards.forEach((card) => {
                const isSelected = card.id === selectedSituationId;
                const el = document.createElement('div');
                el.className = `mini-card ${isSelected ? 'selected' : ''}`;
                el.onclick = () => selectSituation(card);
                el.innerHTML = `
                    <div>
                        <div class="mini-card-title">${card.title}</div>
                        <div class="mini-card-desc">${card.desc}</div>
                    </div>
                    <div class="mini-card-action">${isSelected ? '✓ เลือกแล้ว' : 'เลือกไพ่นี้ ✦'}</div>
                `;
                container.appendChild(el);
            });
        }

        function selectSituation(card) {
            selectedSituationId = card.id;
            renderHandCards();

            const displayEl = document.getElementById('selected-situation-display');
            document.getElementById('big-situation-title').innerText = card.title;
            document.getElementById('big-situation-desc').innerText = card.desc;
            document.getElementById('big-situation-img').innerHTML = card.svg;

            displayEl.style.display = 'flex';
            displayEl.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }

        function getRandom(arr) {
            return arr[Math.floor(Math.random() * arr.length)];
        }

        function flipCard(cardId) {
            document.getElementById(cardId).classList.toggle('flipped');
        }

        function drawMainCards() {
            document.getElementById('tarot-card').classList.remove('flipped');
            document.getElementById('retrieval-card').classList.remove('flipped');

            setTimeout(() => {
                const randomTarot = getRandom(tarotCards);
                const randomRetrieval = getRandom(retrievalCards);

                document.getElementById('tarot-title').innerText = randomTarot.title;
                document.getElementById('tarot-desc').innerText = randomTarot.desc;
                document.getElementById('tarot-img').innerHTML = randomTarot.svg;

                document.getElementById('retrieval-title').innerText = randomRetrieval.title;
